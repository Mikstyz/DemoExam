BEGIN;

CREATE TABLE categories (
    category_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE roles (
    role_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    role_name VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE users (
    user_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    last_name VARCHAR(100) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    middle_name VARCHAR(100),
    login VARCHAR(100) NOT NULL UNIQUE,
    role_id INTEGER NOT NULL REFERENCES roles(role_id)
);

CREATE TABLE subcategories (
    subcategory_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    category_id INTEGER NOT NULL REFERENCES categories(category_id),
    CONSTRAINT uq_subcategory_category_name UNIQUE (category_id, name)
);

CREATE TABLE sizes (
    size_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    size_value VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE products (
    product_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    category_id INTEGER NOT NULL REFERENCES categories(category_id),
    subcategory_id INTEGER REFERENCES subcategories(subcategory_id),
    image_path VARCHAR(500),
    product_name VARCHAR(200) NOT NULL,
    manufacturer VARCHAR(150) NOT NULL,
    description VARCHAR(2000),
    composition VARCHAR(1000),
    price NUMERIC(10, 2) NOT NULL CHECK (price > 0)
);

CREATE TABLE stock_items (
    stock_item_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    product_id INTEGER NOT NULL REFERENCES products(product_id),
    size_id INTEGER NOT NULL REFERENCES sizes(size_id),
    quantity_available INTEGER NOT NULL DEFAULT 0
        CHECK (quantity_available >= 0),
    CONSTRAINT uq_stock_product_size UNIQUE (product_id, size_id)
);

CREATE TABLE orders (
    order_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    order_date TIMESTAMP WITHOUT TIME ZONE NOT NULL,
    user_id INTEGER NOT NULL REFERENCES users(user_id)
);

CREATE TABLE order_items (
    order_item_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    order_id INTEGER NOT NULL REFERENCES orders(order_id)
        ON DELETE CASCADE,
    product_id INTEGER NOT NULL REFERENCES products(product_id),
    size_id INTEGER NOT NULL REFERENCES sizes(size_id),
    quantity INTEGER NOT NULL CHECK (quantity > 0),
    price NUMERIC(10, 2) NOT NULL CHECK (price >= 0)
);

CREATE INDEX ix_users_login ON users(login);
CREATE INDEX ix_users_role_id ON users(role_id);
CREATE INDEX ix_categories_name ON categories(name);
CREATE INDEX ix_roles_role_name ON roles(role_name);
CREATE INDEX ix_subcategories_name ON subcategories(name);
CREATE INDEX ix_subcategories_category_id ON subcategories(category_id);
CREATE INDEX ix_products_category_id ON products(category_id);
CREATE INDEX ix_products_subcategory_id ON products(subcategory_id);
CREATE INDEX ix_products_product_name ON products(product_name);
CREATE INDEX ix_stock_items_product_id ON stock_items(product_id);
CREATE INDEX ix_stock_items_size_id ON stock_items(size_id);
CREATE INDEX ix_orders_order_date ON orders(order_date);
CREATE INDEX ix_orders_user_id ON orders(user_id);
CREATE INDEX ix_order_items_order_id ON order_items(order_id);
CREATE INDEX ix_order_items_product_id ON order_items(product_id);
CREATE INDEX ix_order_items_size_id ON order_items(size_id);

COMMIT;
